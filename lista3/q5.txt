float somaDistancias(Ponto *origem, int n) {
    // também, ninguém checou se origem == NULL.
    if(origem == NULL){
        return -1.0;
    }
    
    float s = 0.0f;
    for (int i = 0; i &lt; n; i++) {
        Ponto *p = criarPonto((float) i, (float) i); // erro 1: estamos sobrescrevendo o ponteiro a cada iteração.
        // também não há uma checagem se o retorno do ponteiro foi NULL.
        if (p == NULL){
            printf("não há memoria disponivel");
            return -1.0;
        }
        s += distancia(p, origem);
        // adição:
        liberarPonto(p); 
    }
    *remover* -> liberarPonto(origem);
    - // observação, nos recebemos origem como um ponteiro, o que significa que essa função não criou este ponteiro
    liberá-lo dentro dessa função pode causar um double free na main.
    - // erro2: nós não liberamos nenhum dos ponteiros que criamos previamente. portanto temos um memory leak.
    return s; (removido: -> distancia(origem, origem)) //use after free, normalmente é comportamento inesperado.
    - // erro 3: como liberamos o ponto, não há como calcular a distancia de um ponto que não exite mais. (neste caso, o retorno é -1)
}
