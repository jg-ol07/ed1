            insere_Pilha(pi, s[i]);
        }else if(s[i] == ')' || s[i] == ']' || s[i] == '}'){

                libera_Pilha(pi);
                return 0;

                printf("WARNING: nao ta aninhado\n");
                libera_Pilha(pi);
                return 0;
            }

            remove_Pilha(pi);
        }

    }

    int valida = Pilha_vazia(pi);
    libera_Pilha(pi);
